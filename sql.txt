CREATE DATABASE IF NOT EXISTS web_app;

USE web_app;

CREATE TABLE IF NOT EXISTS Users (
    username VARCHAR(50) PRIMARY KEY,
    password_hash VARCHAR(255) NOT NULL,
    firstname VARCHAR(50) NOT NULL,
    lastname VARCHAR(50) NOT NULL,
    salary DECIMAL(12,2),
    age INT,
    registerday DATE NOT NULL,
    signintime DATETIME DEFAULT NULL
);
-- 1. Search by first and/or last name
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE firstname LIKE '%John%'
   OR lastname LIKE '%Smith%';

-- 2. Search by User ID
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE username = 'john';

-- 3. Search by salary range
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE salary BETWEEN 50000 AND 70000;

-- 4. Search by age range
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE age BETWEEN 20 AND 30;

-- 5. Users registered after a specified user
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday > (
    SELECT registerday FROM Users WHERE username = 'john'
);

-- 6. Users who have never signed in
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE signintime IS NULL;

-- 7. Users registered on the same day as a specified user
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday = (
    SELECT registerday FROM Users WHERE username = 'john'
);

-- 8. Users registered today
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday = CURDATE();
